# Ver3 CAD pilot - pinned toolchain.
#
# Exact versions resolved and verified on the CAD-1 pilot host
# (see TOOLCHAIN_LOCK.yaml for the platform and the capability probes).
#
# Python 3.8.x is required: cadquery-ocp 7.7.0 publishes cp38 wheels, and the
# pinned resolution below was verified only on that interpreter.
#
# Install with tools/create_env.sh, or directly:
#   python3.8 -m venv <env>
#   <env>/bin/pip install -r requirements-cad.txt

cadquery==2.4.0
cadquery-ocp==7.7.0

# cadquery 2.4.0 dependencies that pip does not pin tightly enough to reproduce.
# multimethod must be 1.9.1: 2.4.0 declares ==1.9.1 and later releases change
# dispatch behaviour. `path` is required by cadquery but is not installed
# transitively by every pip version.
multimethod==1.9.1
path==17.0.0
typish==1.9.3
nptyping==2.0.1
nlopt==2.7.1
casadi==3.7.2
ezdxf==1.1.4

numpy==1.24.4
scipy==1.10.1

# Validation and rendering support.
PyYAML==6.0.2
matplotlib==3.7.5
pillow==10.4.0

# Rigid-body operating simulation for EXE-BM001-01. 2.3.7 is the last release
# that publishes cp38 wheels.
mujoco==2.3.7

# Review-video encoding. imageio-ffmpeg ships a static ffmpeg 4.2.2 with
# libx264, so H.264 output does not depend on whatever ffmpeg the host happens
# to have - and on this host it had none.
imageio==2.35.1
imageio-ffmpeg==0.5.1
